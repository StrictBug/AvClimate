# To start server
source .venv/bin/activate
python -m uvicorn webapp.backend.main:app --host 0.0.0.0 --port 8000

# Auto reload during development
python -m uvicorn webapp.backend.main:app --reload --reload-dir webapp --host 0.0.0.0 --port 8000

#If it says url is already in use
lsof -ti :8000 | xargs -r kill

#push updates to main
git add .
git commit -m "Updates"
git push origin main
